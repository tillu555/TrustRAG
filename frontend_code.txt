========== APP.TSX ==========
import { useEffect, useRef, useState } from "react";
import {
  Activity, BookOpen, Bot, CheckCircle2, ChevronRight,
  FileText, FolderOpen, LayoutDashboard, Loader2,
  MessageSquare, Plus, Search, Send, ShieldCheck,
  UploadCloud, X, AlertCircle
} from "lucide-react";
import "./App.css";

const API = "http://127.0.0.1:8000";

type DocumentItem = {
  document_id: string;
  file_name: string;
  page_count: number;
  section_count: number;
  chunk_count: number;
};

type Source = {
  source_id: string;
  document_name: string;
  section: string;
  page_numbers: number[];
  chunk_id?: string;
};

type Message = {
  role: "user" | "assistant";
  content: string;
  sources?: Source[];
};

function App() {
  const [documents, setDocuments] = useState<DocumentItem[]>([]);
  const [question, setQuestion] = useState("");
  const [messages, setMessages] = useState<Message[]>([]);
  const [uploading, setUploading] = useState(false);
  const [asking, setAsking] = useState(false);
  const [error, setError] = useState("");
  const [apiOnline, setApiOnline] = useState(false);
  const fileInput = useRef<HTMLInputElement>(null);
  const chatEnd = useRef<HTMLDivElement>(null);

  async function loadDocuments() {
    try {
      const response = await fetch(`${API}/documents`);
      if (!response.ok) throw new Error("Could not load library");
      const data = await response.json();
      setDocuments(data.documents ?? []);
      setApiOnline(true);
      setError("");
    } catch {
      setApiOnline(false);
      setError("Cannot connect to TrustRAG API. Check that FastAPI is running on port 8000.");
    }
  }

  useEffect(() => {
    loadDocuments();
  }, []);

  useEffect(() => {
    chatEnd.current?.scrollIntoView({ behavior: "smooth" });
  }, [messages, asking]);

  async function uploadPDF(file?: File) {
    if (!file) return;

    if (!file.name.toLowerCase().endsWith(".pdf")) {
      setError("Please select a PDF file.");
      return;
    }

    setUploading(true);
    setError("");

    const formData = new FormData();
    formData.append("file", file);

    try {
      const response = await fetch(`${API}/documents/upload`, {
        method: "POST",
        body: formData,
      });
      const data = await response.json();

      if (!response.ok) {
        throw new Error(data.detail || "PDF upload failed.");
      }

      await loadDocuments();
    } catch (err) {
      setError(err instanceof Error ? err.message : "Upload failed.");
    } finally {
      setUploading(false);
      if (fileInput.current) fileInput.current.value = "";
    }
  }

  async function askQuestion(event: React.FormEvent) {
    event.preventDefault();
    const currentQuestion = question.trim();

    if (!currentQuestion || asking) return;

    if (documents.length === 0) {
      setError("Upload a research paper before asking a question.");
      return;
    }

    setError("");
    setQuestion("");
    setMessages((previous) => [
      ...previous,
      { role: "user", content: currentQuestion },
    ]);
    setAsking(true);

    try {
      const response = await fetch(`${API}/query`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          question: currentQuestion,
          top_k: 5,
          candidate_k: 30,
        }),
      });

      const data = await response.json();

      if (!response.ok) {
        throw new Error(data.detail || "Question request failed.");
      }

      setMessages((previous) => [
        ...previous,
        {
          role: "assistant",
          content: data.answer,
          sources: data.sources ?? [],
        },
      ]);
    } catch (err) {
      setMessages((previous) => [
        ...previous,
        {
          role: "assistant",
          content: err instanceof Error ? err.message : "Something went wrong.",
        },
      ]);
    } finally {
      setAsking(false);
    }
  }

  const totalPages = documents.reduce((sum, doc) => sum + doc.page_count, 0);
  const totalChunks = documents.reduce((sum, doc) => sum + doc.chunk_count, 0);

  return (
    <div className="app-shell">
      <aside className="sidebar">
        <div className="brand">
          <div className="brand-mark"><ShieldCheck size={23} /></div>
          <div>
            <div className="brand-name">TrustRAG</div>
            <div className="brand-caption">RESEARCH INTELLIGENCE</div>
          </div>
        </div>

        <div className="nav-label">WORKSPACE</div>
        <nav className="navigation">
          <button className="nav-item active">
            <LayoutDashboard size={18} /> Overview
          </button>
          <button className="nav-item" onClick={() => fileInput.current?.click()}>
            <FolderOpen size={18} /> Research library
          </button>
          <button className="nav-item" onClick={() => document.getElementById("assistant")?.scrollIntoView({ behavior: "smooth" })}>
            <MessageSquare size={18} /> AI assistant
          </button>
          <button className="nav-item" title="Evaluation dashboard is planned">
            <Activity size={18} /> Evaluation
            <span className="nav-soon">SOON</span>
          </button>
        </nav>

        <div className="sidebar-bottom">
          <div className="system-card">
            <div className="system-heading">
              <span className={`status-dot ${apiOnline ? "online" : ""}`} />
              System status
            </div>
            <p>{apiOnline ? "API connected" : "API disconnected"}</p>
            <span className="system-subtitle">Local research environment</span>
          </div>
          <div className="sidebar-footer">TrustRAG · Prototype v0.1</div>
        </div>
      </aside>

      <main className="main-area">
        <header className="topbar">
          <div className="breadcrumb">Workspace <ChevronRight size={14} /> Overview</div>
          <div className="topbar-right">
            <span className={`connection ${apiOnline ? "connected" : ""}`}>
              <span className="status-dot" />
              {apiOnline ? "Backend connected" : "Backend offline"}
            </span>
            <div className="avatar">T</div>
          </div>
        </header>

        <div className="content">
          <section className="welcome-row">
            <div>
              <div className="eyebrow">RESEARCH WORKSPACE</div>
              <h1>Good research starts with <span>trusted evidence.</span></h1>
              <p>Explore scientific papers, retrieve relevant passages and ask evidence-grounded questions.</p>
            </div>
            <button className="primary-button" onClick={() => fileInput.current?.click()} disabled={uploading}>
              <Plus size={17} /> Add research paper
            </button>
          </section>

          {error && (
            <div className="error-banner">
              <AlertCircle size={17} />
              <span>{error}</span>
              <button onClick={() => setError("")}><X size={16} /></button>
            </div>
          )}

          <section className="stats-grid">
            <div className="stat-card">
              <div className="stat-icon purple"><BookOpen size={20} /></div>
              <div className="stat-label">Research papers</div>
              <div className="stat-value">{documents.length}</div>
              <div className="stat-foot">Indexed in your library</div>
            </div>
            <div className="stat-card">
              <div className="stat-icon blue"><FileText size={20} /></div>
              <div className="stat-label">Total pages</div>
              <div className="stat-value">{totalPages}</div>
              <div className="stat-foot">Across indexed papers</div>
            </div>
            <div className="stat-card">
              <div className="stat-icon green"><Search size={20} /></div>
              <div className="stat-label">Evidence chunks</div>
              <div className="stat-value">{totalChunks}</div>
              <div className="stat-foot">Available for retrieval</div>
            </div>
            <div className="stat-card">
              <div className="stat-icon amber"><ShieldCheck size={20} /></div>
              <div className="stat-label">Verification</div>
              <div className="stat-value status-text">Not enabled</div>
              <div className="stat-foot">Claim verification is in development</div>
            </div>
          </section>

          <section className="workspace-grid">
            <div className="panel library-panel" id="library">
              <div className="panel-header">
                <div>
                  <h2>Research library</h2>
                  <p>Your indexed scientific documents</p>
                </div>
                <button className="icon-button" onClick={loadDocuments} title="Refresh library">
                  <Activity size={17} />
                </button>
              </div>

              <div
                className="upload-zone"
                onClick={() => fileInput.current?.click()}
                onDragOver={(event) => event.preventDefault()}
                onDrop={(event) => {
                  event.preventDefault();
                  uploadPDF(event.dataTransfer.files[0]);
                }}
              >
                {uploading ? <Loader2 className="spin" size={27} /> : <UploadCloud size={27} />}
                <strong>{uploading ? "Indexing your paper..." : "Upload a research paper"}</strong>
                <span>Drag and drop a PDF here, or browse files</span>
                <small>PDF · Maximum 50 MB</small>
              </div>

              <div className="document-list">
                <div className="list-heading">
                  <span>DOCUMENTS</span>
                  <span>{documents.length} total</span>
                </div>

                {documents.length === 0 ? (
                  <div className="empty-state">
                    <div className="empty-icon"><FileText size={24} /></div>
                    <strong>Your library is empty</strong>
                    <p>Upload your first scientific paper to start exploring its evidence.</p>
                  </div>
                ) : documents.map((doc) => (
                  <div className="document-row" key={doc.document_id}>
                    <div className="pdf-icon">PDF</div>
                    <div className="document-info">
                      <strong title={doc.file_name}>{doc.file_name}</strong>
                      <span>{doc.page_count} pages · {doc.section_count} sections · {doc.chunk_count} chunks</span>
                    </div>
                    <CheckCircle2 className="indexed-icon" size={18} />
                  </div>
                ))}
              </div>
            </div>

            <div className="panel assistant-panel" id="assistant">
              <div className="panel-header assistant-heading">
                <div className="assistant-title">
                  <div className="assistant-icon"><Bot size={20} /></div>
                  <div>
                    <h2>Research assistant</h2>
                    <p>Ask questions across your indexed papers</p>
                  </div>
                </div>
                <span className="model-chip">LOCAL RAG</span>
              </div>

              <div className="chat-area">
                {messages.length === 0 ? (
                  <div className="chat-empty">
                    <div className="chat-orb"><Bot size={27} /></div>
                    <h3>What would you like to investigate?</h3>
                    <p>Ask a question and TrustRAG will retrieve relevant passages from your research library.</p>
                    <div className="suggested-questions">
                      {[
                        "What is the main contribution of these papers?",
                        "Which methods do the authors use?",
                        "What limitations do the papers report?",
                      ].map((item) => (
                        <button key={item} onClick={() => setQuestion(item)}>
                          {item}<ChevronRight size={15} />
                        </button>
                      ))}
                    </div>
                  </div>
                ) : (
                  <div className="messages">
                    {messages.map((message, index) => (
                      <div className={`message ${message.role}`} key={index}>
                        <div className="message-avatar">
                          {message.role === "assistant" ? <Bot size={17} /> : "T"}
                        </div>
                        <div className="message-body">
                          <div className="message-role">{message.role === "assistant" ? "TrustRAG" : "You"}</div>
                          <p>{message.content}</p>
                          {message.sources && message.sources.length > 0 && (
                            <div className="source-list">
                              <div className="source-heading">RETRIEVED SOURCES</div>
                              {message.sources.map((source) => (
                                <div className="source-item" key={source.source_id}>
                                  <span className="source-id">{source.source_id}</span>
                                  <div>
                                    <strong>{source.document_name}</strong>
                                    <span>{source.section} · Pages {source.page_numbers?.join(", ") || "Not available"}</span>
                                  </div>
                                </div>
                              ))}
                            </div>
                          )}
                        </div>
                      </div>
                    ))}
                    {asking && (
                      <div className="thinking"><Loader2 className="spin" size={17} /> Retrieving evidence and generating answer...</div>
                    )}
                    <div ref={chatEnd} />
                  </div>
                )}
              </div>

              <form className="chat-composer" onSubmit={askQuestion}>
                <input
                  value={question}
                  onChange={(event) => setQuestion(event.target.value)}
                  placeholder={documents.length ? "Ask a question about your research..." : "Upload a paper to start asking questions"}
                  disabled={documents.length === 0 || asking}
                />
                <button type="submit" disabled={!question.trim() || asking || documents.length === 0}>
                  {asking ? <Loader2 className="spin" size={18} /> : <Send size={18} />}
                </button>
              </form>
              <div className="assistant-disclaimer">Answers are generated from retrieved evidence. Independent claim verification is not yet enabled.</div>
            </div>
          </section>

          <footer className="page-footer">
            <span>TrustRAG Research Prototype</span>
            <span><span className="status-dot online" /> Local-first · Your documents stay on this machine</span>
          </footer>
        </div>
      </main>

      <input
        ref={fileInput}
        type="file"
        accept=".pdf,application/pdf"
        hidden
        onChange={(event) => uploadPDF(event.target.files?.[0])}
      />
    </div>
  );
}

export default App;
